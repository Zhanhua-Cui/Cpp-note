## B1. RAII 计时器

```cpp
#include <iostream>
#include <chrono>

class Timer {
    std::chrono::steady_clock::time_point start_;
public:
    Timer() : start_(std::chrono::steady_clock::now()) {}

    ~Timer() {
        auto end = std::chrono::steady_clock::now();
        auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(end - start_).count();
        std::cout << "耗时 " << ms << " 毫秒\n";
    }
};

int main() {
    {
        Timer t;
        // 一段要测的代码，比如空转
        volatile long long sum = 0;
        for (long long i = 0; i < 100000000; ++i) sum += i;
    }   // t 析构，自动打印耗时
}
```

**要点**：计时开始放构造、打印放析构，就是 RAII——无论作用域怎么退出都会执行析构。

## B2. 五法则 + 触发场景

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <utility>

class Widget {
    std::string name_;
    std::vector<int> data_;
public:
    explicit Widget(std::string n) : name_(std::move(n)), data_(1000, 42) {}

    ~Widget()                        { std::cout << "析构: " << name_ << "\n"; }
    Widget(const Widget& o)          : name_(o.name_), data_(o.data_)
                                     { std::cout << "拷贝构造: " << name_ << "\n"; }
    Widget(Widget&& o) noexcept      : name_(std::move(o.name_)), data_(std::move(o.data_))
                                     { std::cout << "移动构造: " << name_ << "\n"; }
    Widget& operator=(const Widget& o) {
        std::cout << "拷贝赋值: " << name_ << "\n";
        name_ = o.name_; data_ = o.data_; return *this;
    }
    Widget& operator=(Widget&& o) noexcept {
        std::cout << "移动赋值: " << name_ << "\n";
        name_ = std::move(o.name_); data_ = std::move(o.data_); return *this;
    }
};

int main() {
    Widget a("a");
    Widget b(a);                // 拷贝构造
    Widget c(std::move(a));     // 移动构造
    Widget d("d");
    d = b;                      // 拷贝赋值
    d = std::move(c);           // 移动赋值
}
```

**要点**：`Widget` 成员都是值类型（string/vector），其实可以走「零法则」什么都不写；这里写出来只是为了观察每个函数何时被调用。

## B3. 智能指针

```cpp
#include <iostream>
#include <memory>

struct X { int v; ~X() { std::cout << "X 析构\n"; } };

int main() {
    auto u = std::make_unique<X>();
    // auto u2 = u;              // ❌ unique_ptr 不可拷贝（独占所有权）
    auto u2 = std::move(u);      // ✅ 所有权转移

    auto s1 = std::make_shared<X>();
    std::cout << "s1.count=" << s1.use_count() << "\n";   // 1
    {
        auto s2 = s1;
        std::cout << "s1.count=" << s1.use_count() << "\n"; // 2
    }
    std::cout << "s1.count=" << s1.use_count() << "\n";   // 1
}
```

**要点**：`unique_ptr` 不可拷贝因为它独占所有权，靠移动转移；`shared_ptr` 靠引用计数，计数归零才释放。

## A1. MyString（五法则）

```cpp
#include <iostream>
#include <cstring>
#include <utility>

class MyString {
    char* data_;
    size_t size_;
public:
    explicit MyString(const char* s = "") : size_(strlen(s)) {
        data_ = new char[size_ + 1];
        strcpy(data_, s);
    }

    ~MyString() { delete[] data_; }

    // 拷贝构造
    MyString(const MyString& o) : size_(o.size_) {
        data_ = new char[size_ + 1];
        strcpy(data_, o.data_);
    }

    // 移动构造
    MyString(MyString&& o) noexcept : data_(o.data_), size_(o.size_) {
        o.data_ = nullptr;
        o.size_ = 0;
    }

    // 拷贝赋值
    MyString& operator=(const MyString& o) {
        if (this == &o) return *this;
        char* tmp = new char[o.size_ + 1];
        strcpy(tmp, o.data_);
        delete[] data_;
        data_ = tmp;
        size_ = o.size_;
        return *this;
    }

    // 移动赋值
    MyString& operator=(MyString&& o) noexcept {
        if (this == &o) return *this;
        delete[] data_;
        data_ = o.data_;
        size_ = o.size_;
        o.data_ = nullptr;
        o.size_ = 0;
        return *this;
    }

    void append(const char* s) {
        size_t add = strlen(s);
        char* tmp = new char[size_ + add + 1];
        strcpy(tmp, data_);
        strcat(tmp, s);
        delete[] data_;
        data_ = tmp;
        size_ += add;
    }

    const char* c_str() const { return data_; }
    size_t size() const { return size_; }
};

int main() {
    MyString a("hello");
    MyString b = a;                 // 拷贝
    MyString c = std::move(a);      // 移动
    std::cout << "b=" << b.c_str() << " c=" << c.c_str()
              << " a.size=" << a.size() << "（a 已被掏空）\n";
}
```

**要点**：拷贝赋值用「先分配临时再释放旧的」或「先释放再分配」都行，关键是自赋值保护和移动里置空 `o.data_`。

## A2. 打破循环引用

```cpp
#include <iostream>
#include <memory>

struct Node {
    std::shared_ptr<Node> next;
    std::weak_ptr<Node>   prev;   // 弱引用打破环
    int id;
    explicit Node(int i) : id(i) {}
    ~Node() { std::cout << "Node " << id << " 析构\n"; }
};

int main() {
    auto head = std::make_shared<Node>(1);
    auto tail = std::make_shared<Node>(2);
    head->next = tail;
    tail->prev = head;             // 弱引用不占计数
    // 离开作用域：head、tail 计数都归零，正常释放
}
```

**要点**：环中至少一处用 `weak_ptr`，`prev` 方向是天然的弱引用选择。

## A3. 容器 + 算法

```cpp
#include <algorithm>
#include <numeric>
#include <vector>
#include <set>
#include <iostream>

int main() {
    std::vector<int> v{150, 20, 300, 20, 150, 500, 80};

    // 1. 去重并排序
    std::set<int> s(v.begin(), v.end());          // set 自动去重+有序
    std::vector<int> unique_sorted(s.begin(), s.end());

    // 2. 统计 >100 的个数
    long big = std::count_if(v.begin(), v.end(), [](int x) { return x > 100; });

    // 3. 每个元素 *3
    std::vector<int> tripled(v.size());
    std::transform(v.begin(), v.end(), tripled.begin(), [](int x) { return x * 3; });

    // 4. 平均值
    double avg = std::accumulate(v.begin(), v.end(), 0.0) / v.size();

    std::cout << "big=" << big << " avg=" << avg << "\n";
    std::cout << "tripled: ";
    for (int x : tripled) std::cout << x << ' ';
    std::cout << "\n";
}
```

**要点**：用 `set` 去重最省事；`accumulate` 的初值用 `0.0` 保证是浮点累加。

## A4. lambda 捕获

```cpp
#include <iostream>
#include <functional>

std::function<int()> make_counter(int start) {
    int n = start;                       // 局部变量
    return [n]() mutable { return n++; };// 按值捕获 + mutable
    // ❌ 若按引用捕获 [&n]，函数返回后 n 已销毁，悬垂引用 → 未定义行为
}

int main() {
    auto c = make_counter(10);
    std::cout << c() << ' ' << c() << ' ' << c() << '\n';  // 10 11 12
}
```

**要点**：返回 lambda 时，局部变量必须**按值捕获**（拷走），不能按引用捕获（会悬垂）。`mutable` 允许在 lambda 内修改按值捕获的副本。

## C1. 自定义 MyVector（移动语义 + 模板）

```cpp
#include <iostream>
#include <string>
#include <utility>

struct Heavy {
    std::string s;
    Heavy() = default;                                     // 默认构造：new T[n] 需要它
    explicit Heavy(std::string str) : s(std::move(str)) {}
    Heavy(const Heavy& o)            : s(o.s)             { std::cout << "  Heavy 拷贝\n"; }
    Heavy(Heavy&& o) noexcept        : s(std::move(o.s))  { std::cout << "  Heavy 移动\n"; }
    Heavy& operator=(const Heavy& o) { s = o.s;           std::cout << "  Heavy 拷贝赋值\n"; return *this; }
    Heavy& operator=(Heavy&& o) noexcept { s = std::move(o.s); std::cout << "  Heavy 移动赋值\n"; return *this; }
};

template <typename T>
class MyVector {
    T* data_;
    size_t size_;
    size_t cap_;
public:
    MyVector() : data_(nullptr), size_(0), cap_(0) {}

    ~MyVector() { delete[] data_; }

    MyVector(const MyVector& o) : size_(o.size_), cap_(o.cap_) {
        data_ = new T[cap_];
        for (size_t i = 0; i < size_; ++i) data_[i] = o.data_[i];   // 拷贝
    }

    MyVector(MyVector&& o) noexcept : data_(o.data_), size_(o.size_), cap_(o.cap_) {
        o.data_ = nullptr; o.size_ = o.cap_ = 0;                      // 掏空
    }

    MyVector& operator=(const MyVector& o) {
        if (this == &o) return *this;
        delete[] data_;
        size_ = o.size_; cap_ = o.cap_;
        data_ = new T[cap_];
        for (size_t i = 0; i < size_; ++i) data_[i] = o.data_[i];
        return *this;
    }

    MyVector& operator=(MyVector&& o) noexcept {
        if (this == &o) return *this;
        delete[] data_;
        data_ = o.data_; size_ = o.size_; cap_ = o.cap_;
        o.data_ = nullptr; o.size_ = o.cap_ = 0;
        return *this;
    }

    void push_back(const T& v) {
        if (size_ == cap_) reserve(cap_ ? cap_ * 2 : 1);
        data_[size_++] = v;
    }

    void push_back(T&& v) {
        if (size_ == cap_) reserve(cap_ ? cap_ * 2 : 1);
        data_[size_++] = std::move(v);
    }

    void reserve(size_t n) {
        if (n <= cap_) return;
        T* tmp = new T[n];
        for (size_t i = 0; i < size_; ++i)
            tmp[i] = std::move(data_[i]);    // 扩容搬移：移动而非拷贝
        delete[] data_;
        data_ = tmp;
        cap_ = n;
    }

    T& operator[](size_t i) { return data_[i]; }
    size_t size() const { return size_; }
    size_t capacity() const { return cap_; }
};

int main() {
    MyVector<Heavy> v;
    std::cout << "push_back 三次（观察扩容时是移动还是拷贝）:\n";
    v.push_back(Heavy("one"));
    v.push_back(Heavy("two"));
    v.push_back(Heavy("three"));
    // 每次扩容 reserve 里都是 std::move 搬移 → 打印 "Heavy 移动"
}
```

**要点**：扩容搬移必须用 `std::move(data_[i])`；移动构造/赋值 `noexcept` 且掏空源对象。

> **一个容易踩的坑**：上面 `reserve` 里 `new T[n]` 要求 `T` 是**可默认构造**的（所以 `Heavy` 里写了 `Heavy() = default;`）。真正的 `std::vector` 用 `allocator` + placement new 来避免这个限制——元素类型可以不默认构造。你自己写简版知道这个区别即可。

## C2. 简化版 MySharedPtr

```cpp
#include <iostream>

template <typename T>
class MySharedPtr {
    T* ptr_;
    long* ref_;
public:
    explicit MySharedPtr(T* p = nullptr) : ptr_(p), ref_(new long(1)) {}

    MySharedPtr(const MySharedPtr& o) : ptr_(o.ptr_), ref_(o.ref_) { ++(*ref_); }

    MySharedPtr(MySharedPtr&& o) noexcept : ptr_(o.ptr_), ref_(o.ref_) {
        o.ptr_ = nullptr; o.ref_ = nullptr;
    }

    ~MySharedPtr() { release(); }

    MySharedPtr& operator=(const MySharedPtr& o) {
        if (this == &o) return *this;
        release();
        ptr_ = o.ptr_; ref_ = o.ref_;
        ++(*ref_);
        return *this;
    }

    T& operator*()  const { return *ptr_; }
    T* operator->() const { return ptr_; }
    long use_count() const { return ref_ ? *ref_ : 0; }

private:
    void release() {
        if (ref_ && --(*ref_) == 0) {
            delete ptr_;
            delete ref_;
        }
    }
};

struct Foo { int x = 7; };

int main() {
    MySharedPtr<Foo> p1(new Foo);
    {
        MySharedPtr<Foo> p2 = p1;
        std::cout << "count=" << p1.use_count() << "\n";   // 2
    }
    std::cout << "count=" << p1.use_count() << "\n";       // 1
    std::cout << "x=" << p1->x << "\n";
}
```

**要点**：核心是「控制块」——一个独立的 `long* ref_` 被所有副本共享，拷贝时 +1，析构时 -1 归零则释放对象和控制块。真实 `shared_ptr` 用原子操作和虚删除器，这里简化了。
